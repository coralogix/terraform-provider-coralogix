func decodeAlert(t *testing.T, js string) alerts.AlertDef {
	t.Helper()
	var a alerts.AlertDef
	if err := json.Unmarshal([]byte(js), &a); err != nil {
		t.Fatalf("%v\n%s", err, js)
	}
	return a
}

func alertSchemas() (hw, switched schema.Schema) {
	var resp resource.SchemaResponse
	(&AlertResource{}).Schema(context.Background(), resource.SchemaRequest{}, &resp)
	return resp.Schema, schema.Schema{Version: resp.Schema.Version, Attributes: switchedAlertAttributes()}
}

// hwAlertState is the state of the handwritten flatten of the alert, with
// the current schedule and notification group of cur (nil for an import).
func hwAlertState(t *testing.T, s schema.Schema, a alerts.AlertDef, cur *alerttypes.AlertResourceModel) tftypes.Value {
	t.Helper()
	schedule, group := types.ObjectNull(nil), types.ObjectNull(nil)
	if cur != nil {
		schedule, group = cur.Schedule, cur.NotificationGroup
	}
	return equivalenceState(t, s, func(st *tfsdk.State) diag.Diagnostics {
		m, diags := flattenAlert(context.Background(), a, &schedule, &group)
		if diags.HasError() {
			return diags
		}
		return st.Set(context.Background(), m)
	})
}

func swAlertState(t *testing.T, s schema.Schema, a alerts.AlertDef, cur *switchedAlertModel) tftypes.Value {
	t.Helper()
	var schedule, group *types.Object
	if cur != nil {
		schedule, group = &cur.Schedule, &cur.NotificationGroup
	}
	return equivalenceState(t, s, func(st *tfsdk.State) diag.Diagnostics {
		m, diags := flattenSwitchedAlert(context.Background(), a, schedule, group)
		if diags.HasError() {
			return diags
		}
		return st.Set(context.Background(), m)
	})
}

func TestAlertEquivalenceFlatten(t *testing.T) {
	ctx := context.Background()
	hwSchema, swSchema := alertSchemas()
	for name, js := range alertCases() {
		t.Run(name, func(t *testing.T) {
			a := decodeAlert(t, js)
			hwImport, swImport := hwAlertState(t, hwSchema, a, nil), swAlertState(t, swSchema, a, nil)
			compareStates(t, "import", hwImport, swImport)
			// A read after the import: the current state holds the
			// schedule time zone and the retriggering nulls.
			var hwCur alerttypes.AlertResourceModel
			if diags := (tfsdk.State{Schema: hwSchema, Raw: hwImport}).Get(ctx, &hwCur); diags.HasError() {
				t.Fatal(diags)
			}
			var swCur switchedAlertModel
			if diags := (tfsdk.State{Schema: swSchema, Raw: swImport}).Get(ctx, &swCur); diags.HasError() {
				t.Fatal(diags)
			}
			compareStates(t, "read", hwAlertState(t, hwSchema, a, &hwCur), swAlertState(t, swSchema, a, &swCur))
		})
	}
}

func compareStates(t *testing.T, step string, hw, sw tftypes.Value) {
	t.Helper()
	diffs, err := hw.Diff(sw)
	if err != nil {
		t.Fatal(err)
	}
	for _, d := range diffs {
		t.Errorf("%s: state differs at %s: handwritten %v, switched %v", step, d.Path, d.Value1, d.Value2)
	}
}

func TestAlertEquivalenceExpand(t *testing.T) {
	ctx := context.Background()
	hwSchema, swSchema := alertSchemas()
	for name, js := range alertCases() {
		t.Run(name, func(t *testing.T) {
			raw := hwAlertState(t, hwSchema, decodeAlert(t, js), nil)
			var hwModel alerttypes.AlertResourceModel
			if diags := (tfsdk.State{Schema: hwSchema, Raw: raw}).Get(ctx, &hwModel); diags.HasError() {
				t.Fatal(diags)
			}
			var swModel switchedAlertModel
			if diags := (tfsdk.State{Schema: swSchema, Raw: raw}).Get(ctx, &swModel); diags.HasError() {
				t.Fatal(diags)
			}
			hwBody, diags := extractAlertProperties(ctx, &hwModel)
			if diags.HasError() {
				t.Fatal(diags)
			}
			swBody, diags := expandSwitchedAlert(ctx, &swModel)
			if diags.HasError() {
				t.Fatal(diags)
			}
			if h, s := equivalenceJSON(t, hwBody), equivalenceJSON(t, swBody); h != s {
				t.Errorf("request differs at %v", jsonDiffs(t, h, s))
			}
		})
	}
}

func equivalenceState(t *testing.T, s schema.Schema, set func(*tfsdk.State) diag.Diagnostics) tftypes.Value {
	t.Helper()
	state := tfsdk.State{Schema: s, Raw: tftypes.NewValue(s.Type().TerraformType(context.Background()), nil)}
	if diags := set(&state); diags.HasError() {
		t.Fatal(fmt.Sprint(diags))
	}
	return state.Raw
}

// equivalenceJSON is the JSON of v without empty arrays: the handwritten
// code sends a null list as [], the generated code leaves it out, and for
// the API (protobuf repeated fields) the two are the same.
func equivalenceJSON(t *testing.T, v any) string {
	t.Helper()
	b, err := json.Marshal(v)
	if err != nil {
		t.Fatal(err)
	}
	var tree any
	if err := json.Unmarshal(b, &tree); err != nil {
		t.Fatal(err)
	}
	b, err = json.Marshal(dropEmptyArrays(tree))
	if err != nil {
		t.Fatal(err)
	}
	return string(b)
}

// jsonDiffs lists the paths where the JSON texts a and b differ, with both
// values.
func jsonDiffs(t *testing.T, a, b string) []string {
	t.Helper()
	var x, y any
	if err := json.Unmarshal([]byte(a), &x); err != nil {
		t.Fatal(err)
	}
	if err := json.Unmarshal([]byte(b), &y); err != nil {
		t.Fatal(err)
	}
	var out []string
	var walk func(p string, x, y any)
	walk = func(p string, x, y any) {
		xm, xok := x.(map[string]any)
		ym, yok := y.(map[string]any)
		xl, xlok := x.([]any)
		yl, ylok := y.([]any)
		switch {
		case xok && yok:
			keys := map[string]bool{}
			for k := range xm {
				keys[k] = true
			}
			for k := range ym {
				keys[k] = true
			}
			for k := range keys {
				walk(p+"."+k, xm[k], ym[k])
			}
		case xlok && ylok && len(xl) == len(yl):
			for i := range xl {
				walk(fmt.Sprintf("%s[%d]", p, i), xl[i], yl[i])
			}
		default:
			xb, _ := json.Marshal(x)
			yb, _ := json.Marshal(y)
			if string(xb) != string(yb) {
				out = append(out, fmt.Sprintf("%s: handwritten %s, switched %s", p, xb, yb))
			}
		}
	}
	walk("", x, y)
	return out
}

func dropEmptyArrays(v any) any {
	switch x := v.(type) {
	case map[string]any:
		for k, e := range x {
			if a, ok := e.([]any); ok && len(a) == 0 {
				delete(x, k)
				continue
			}
			x[k] = dropEmptyArrays(e)
		}
	case []any:
		for i := range x {
			x[i] = dropEmptyArrays(x[i])
		}
	}
	return v
}
