func sloSchemas() (hw, gen schema.Schema) {
	var resp resource.SchemaResponse
	(&SLOV2Resource{}).Schema(context.Background(), resource.SchemaRequest{}, &resp)
	return resp.Schema, schema.Schema{Version: resp.Schema.Version, Attributes: slotypes.SloAttributes()}
}

func decodeSlo(t *testing.T, js string) *slos.Slo {
	t.Helper()
	var s slos.Slo
	if err := json.Unmarshal([]byte(js), &s); err != nil {
		t.Fatal(err)
	}
	return &s
}

func TestSloEquivalenceFlatten(t *testing.T) {
	ctx := context.Background()
	hwSchema, genSchema := sloSchemas()
	for name, js := range sloFixtures() {
		t.Run(name, func(t *testing.T) {
			hwState := sloState(t, hwSchema, func(s *tfsdk.State) diag.Diagnostics {
				m, diags := flattenSLOV2(ctx, decodeSlo(t, js))
				return append(diags, s.Set(ctx, m)...)
			})
			genState := sloState(t, genSchema, func(s *tfsdk.State) diag.Diagnostics {
				slo := decodeSlo(t, js)
				normalizeSloResponse(slo)
				m, diags := slotypes.FlattenSlo(ctx, path.Empty(), slo)
				return append(diags, s.Set(ctx, m)...)
			})
			diffs, err := hwState.Diff(genState)
			if err != nil {
				t.Fatal(err)
			}
			for _, d := range diffs {
				t.Errorf("state differs at %s: handwritten %v, generated %v", d.Path, d.Value1, d.Value2)
			}
		})
	}
}

func TestSloEquivalenceExpand(t *testing.T) {
	ctx := context.Background()
	hwSchema, genSchema := sloSchemas()
	for name, js := range sloFixtures() {
		t.Run(name, func(t *testing.T) {
			raw := sloState(t, hwSchema, func(s *tfsdk.State) diag.Diagnostics {
				m, diags := flattenSLOV2(ctx, decodeSlo(t, js))
				return append(diags, s.Set(ctx, m)...)
			})
			var hwModel SLOV2ResourceModel
			if diags := (tfsdk.State{Schema: hwSchema, Raw: raw}).Get(ctx, &hwModel); diags.HasError() {
				t.Fatal(diags)
			}
			var genModel slotypes.SloModel
			if diags := (tfsdk.State{Schema: genSchema, Raw: raw}).Get(ctx, &genModel); diags.HasError() {
				t.Fatal(diags)
			}
			hwBody, diags := extractSLOV2(ctx, &hwModel)
			if diags.HasError() {
				t.Fatal(diags)
			}
			genBody, diags := slotypes.ExpandSlo(ctx, path.Empty(), &genModel)
			if diags.HasError() {
				t.Fatal(diags)
			}
			normalizeSloRequest(genBody)
			// The request body is Slo1, a copy of Slo (extractSLOV2Payload).
			h, g := sloJSON(t, extractSLOV2Payload(hwBody)), sloJSON(t, extractSLOV2Payload(genBody))
			if h != g {
				t.Errorf("request differs:\nhandwritten: %s\ngenerated:   %s", h, g)
			}
		})
	}
}

// TestSloEquivalencePlanWithUnknowns checks that an update plan with every computed
// attribute unknown reads into the generated model and expands. The first
// live check found this: apm_sli_metadata and grouping are computed objects.
func TestSloPlanWithUnknowns(t *testing.T) {
	ctx := context.Background()
	_, genSchema := sloSchemas()
	for name, js := range sloFixtures() {
		t.Run(name, func(t *testing.T) {
			// The state of the switched code, so the check also runs with
			// the added fields.
			raw := sloState(t, genSchema, func(s *tfsdk.State) diag.Diagnostics {
				slo := decodeSlo(t, js)
				normalizeSloResponse(slo)
				m, diags := slotypes.FlattenSlo(ctx, path.Empty(), slo)
				return append(diags, s.Set(ctx, m)...)
			})
			plan, err := schemadump.PlanWithUnknowns(ctx, genSchema, raw)
			if err != nil {
				t.Fatal(err)
			}
			var m slotypes.SloModel
			if diags := (tfsdk.Plan{Schema: genSchema, Raw: plan}).Get(ctx, &m); diags.HasError() {
				t.Fatalf("get plan: %v", diags)
			}
			body, diags := slotypes.ExpandSlo(ctx, path.Empty(), &m)
			if diags.HasError() {
				t.Fatalf("expand: %v", diags)
			}
			normalizeSloRequest(body)
		})
	}
}

func sloState(t *testing.T, s schema.Schema, set func(*tfsdk.State) diag.Diagnostics) tftypes.Value {
	t.Helper()
	state := tfsdk.State{Schema: s, Raw: tftypes.NewValue(s.Type().TerraformType(context.Background()), nil)}
	if diags := set(&state); diags.HasError() {
		t.Fatal(fmt.Sprint(diags))
	}
	return state.Raw
}

func sloJSON(t *testing.T, v any) string {
	t.Helper()
	b, err := json.Marshal(v)
	if err != nil {
		t.Fatal(err)
	}
	return string(b)
}
